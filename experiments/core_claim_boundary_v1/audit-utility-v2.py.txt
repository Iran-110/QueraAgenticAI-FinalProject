def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), default=str).encode()


def digest(value):
    return sha(canonical(value))


def connection(path):
    db = sqlite3.connect(path.resolve().as_uri() + '?mode=ro', uri=True)
    db.row_factory = sqlite3.Row
    db.execute('PRAGMA query_only=ON')
    db.execute('BEGIN')
    return db


def request_id(context):
    selected = {key: context[key] for key in ('run_id', 'case_id', 'case_revision', 'attempt_index')}
    return 'call:' + sha(json.dumps(selected, separators=(',', ':')).encode())


def schema(db):
    return [dict(r) for r in db.execute(
        "SELECT type,name,tbl_name,sql FROM sqlite_master WHERE name NOT LIKE 'sqlite_%' ORDER BY type,name"
    )]


def rows(db, table):
    if not re.fullmatch(r'[A-Za-z_][A-Za-z_0-9]*', table):
        raise ValueError('AuditTableRejected')
    return [dict(r) for r in db.execute(f'SELECT rowid AS audit_rowid,* FROM "{table}" ORDER BY rowid')]


def journal(db):
    objects = schema(db)
    return {'schema': objects, 'tables': {
        item['name']: {
            'columns': [r[1] for r in db.execute(f'PRAGMA table_info("{item["name"]}")')],
            'row_sha256': sorted(sha(json.dumps({k: v for k, v in row.items() if k != 'audit_rowid'}, sort_keys=True).encode())
                                 for row in rows(db, item['name']))
        } for item in objects if item['type'] == 'table'
    }}


def check(checks, name, condition, details=None):
    checks.append({'check': name, 'passed': bool(condition), **({'details': details} if details is not None else {})})


def pinned_helpers(source):
    """Import trusted prompt/contract code from the exact recorded Git commit.

    Extraction is temporary and outside Git. No experiment driver, credentials,
    gateway instance, ledger client or index client is created.
    """
    if not re.fullmatch(r'[0-9a-f]{40}', source):
        raise ValueError('AuditSourceCommitRejected')
    if any(name == 'agent_app' or name.startswith('agent_app.') for name in sys.modules):
        raise ValueError('AuditModuleIsolationRejected')
    temporary = tempfile.TemporaryDirectory(prefix='source-order-pinned-audit-', dir=OUT)
    root = Path(temporary.name)
    archived = subprocess.check_output(['git', 'archive', source, 'src/agent_app'], cwd=ROOT)
    with tarfile.open(fileobj=io.BytesIO(archived)) as archive:
        archive.extractall(root, filter='data')
    sys.path.insert(0, str(root / 'src'))
    contracts = importlib.import_module('agent_app.contracts')
    config = importlib.import_module('agent_app.llm.config')
    metis = importlib.import_module('agent_app.llm.metis')
    compact = importlib.import_module('agent_app.reasoning.compact_plan')
    grounding = importlib.import_module('agent_app.reasoning.grounding')
    handoff = importlib.import_module('agent_app.orchestration.handoff')
    capacity = importlib.import_module('agent_app.reasoning.source_capacity')
    selection = importlib.import_module('agent_app.reasoning.source_selection')
    for module in (contracts, config, metis, compact, grounding, handoff, capacity, selection):
        if not Path(module.__file__).resolve().is_relative_to(root / 'src' / 'agent_app'):
            raise ValueError('AuditModuleIsolationRejected')
    return SimpleNamespace(
        temporary=temporary, root=root,
        GroundedRequest=contracts.GroundedRequest, GroundedPlan=contracts.GroundedPlan,
        ProposalHandoff=contracts.ProposalHandoff, validate_handoff=handoff.validate_handoff,
        EvidenceBundle=contracts.EvidenceBundle, EvidenceHit=contracts.EvidenceHit,
        ReasoningContext=contracts.ReasoningContext, MetisConfig=config.MetisConfig,
        configured_grounded_messages=metis.configured_grounded_messages,
        expand_compact_plan=compact.expand_compact_plan,
        render_grounded_plan=grounding.render_grounded_plan,
        GroundingRejected=grounding.GroundingRejected,
        source_content_sha256=grounding.source_content_sha256,
        source_capacity=capacity.source_capacity,
        COMPLEMENTARY_SOURCE_POLICY=selection.COMPLEMENTARY_SOURCE_POLICY,
    )


WIRE_OBSERVATION_FIELDS = frozenset({
    'request_id', 'schema_version', 'wire_format', 'raw_output_sha256',
    'raw_utf8_bytes', 'json_shape', 'decision', 'sources_count',
    'selected_indexes', 'wire_schema_valid', 'canonical_binding_valid',
    'rejection_code', 'unexpected_field_count',
})
DECISIONS = frozenset({'answer', 'ask', 'escalate', 'retrieve_more'})
BINDING_CODES = frozenset({'grounded_context_mismatch', 'grounded_source_mismatch'})


def hash_shape(value):
    return type(value) is str and re.fullmatch(r'[0-9a-f]{64}', value) is not None


def audit_output(checks, reviews, attempt, request, job, send, pinned):
    """Check record consistency without pretending hashes reveal raw JSON.

    Raw provider text is unavailable. Compact successes permit an independent
    current-index expansion and canonical hash check; raw schema flags remain
    explicitly identified as recorded observations, including rejected outputs.
    No raw errors, unknown names, query text or source text are exported.
    """
    rid = request_id(attempt['context'])
    compact = job['compact_plan']
    observation = attempt.get('wire_output')
    review = {
        'request_id': rid, 'arm': job['arm'],
        'canonical_plan_present': attempt.get('plan') is not None,
        'output_sha256_kind': 'locally_expanded_canonical_json' if compact else 'raw_provider_content',
        'raw_wire_schema_independently_reparsed': False,
        'canonical_expansion_independently_checked': False,
        'canonical_output_hash_independently_checked': False,
        'raw_schema_diagnostic_source': 'gateway_raw_wire_observer' if compact else 'raw_provider_plan_observer',
    }
    reviews.append(review)
    if not compact:
        check(checks, 'canonical_arm_preserves_original_observation_shape:' + rid,
              'wire_output' not in attempt)
        observed = attempt.get('schema_observation')
        review['recorded_raw_wire_schema_valid'] = (
            observed.get('syntactic_valid') if type(observed) is dict
            and type(observed.get('syntactic_valid')) is bool else None)
        output_hash = attempt.get('output_sha256')
        if output_hash is not None:
            check(checks, 'canonical_arm_recorded_raw_output_hash_shape:' + rid, hash_shape(output_hash))
            review['recorded_raw_output_sha256'] = output_hash if hash_shape(output_hash) else None
        return
    check(checks, 'compact_arm_raw_observation_field_present:' + rid, 'wire_output' in attempt)
    if observation is None:
        check(checks, 'unobserved_raw_never_claimed_as_valid_or_expanded:' + rid,
              attempt.get('plan') is None and 'schema_observation' not in attempt
              and 'output_sha256' not in attempt)
        review['recorded_raw_wire_schema_valid'] = None
        review['recorded_raw_observation_available'] = False
        if send is not None and send.get('finish_reason') == 'length':
            check(checks, 'length_rejected_before_compact_expansion:' + rid,
                  attempt['error_code'] == 'ProviderResponseInvalid')
        return
    check(checks, 'compact_observation_is_safe_allowlist_only:' + rid,
          type(observation) is dict and set(observation) == WIRE_OBSERVATION_FIELDS)
    if type(observation) is not dict:
        return
    check(checks, 'compact_raw_observation_identity_and_fixed_metadata:' + rid,
          observation.get('request_id') == rid and observation.get('schema_version') == 1
          and observation.get('wire_format') == 'grounded-indexed-plan-v1')
    raw_hash, byte_count = observation.get('raw_output_sha256'), observation.get('raw_utf8_bytes')
    check(checks, 'compact_raw_hash_and_byte_count_shape:' + rid,
          hash_shape(raw_hash) and type(byte_count) is int and 0 < byte_count <= 262144)
    if send is not None:
        check(checks, 'compact_raw_bytes_match_observed_provider_content:' + rid,
              byte_count == send.get('content_utf8_bytes') and send.get('finish_reason') == 'stop')
    wire_valid, binding_valid = observation.get('wire_schema_valid'), observation.get('canonical_binding_valid')
    rejection = observation.get('rejection_code')
    indices, count = observation.get('selected_indexes'), observation.get('sources_count')
    safe_indices = type(indices) is list and len(indices) <= 8 and all(
        type(index) is int and 0 <= index < 100 for index in indices)
    check(checks, 'compact_observation_values_are_bounded_and_allowlisted:' + rid,
          type(wire_valid) is bool and type(binding_valid) is bool
          and observation.get('json_shape') in {'object', 'other', 'invalid'}
          and observation.get('decision') in DECISIONS | {None}
          and rejection in BINDING_CODES | {'invalid_grounded_plan', None}
          and (indices is None or safe_indices)
          and (count is None or type(count) is int and 0 <= count <= 262144)
          and type(observation.get('unexpected_field_count')) is int
          and 0 <= observation['unexpected_field_count'] <= 262144)
    review.update(
        recorded_raw_observation_available=True,
        recorded_raw_output_sha256=raw_hash if hash_shape(raw_hash) else None,
        recorded_raw_utf8_bytes=byte_count if type(byte_count) is int else None,
        recorded_raw_wire_schema_valid=wire_valid if type(wire_valid) is bool else None,
        recorded_canonical_binding_valid=binding_valid if type(binding_valid) is bool else None,
        recorded_rejection_code=rejection if rejection in BINDING_CODES | {'invalid_grounded_plan'} else None,
        selected_indexes=indices if safe_indices else None,
    )
    if not binding_valid:
        check(checks, 'rejected_compact_wire_has_no_local_canonical_output:' + rid,
              attempt.get('plan') is None and 'schema_observation' not in attempt
              and 'output_sha256' not in attempt
              and (wire_valid is False and rejection == 'invalid_grounded_plan'
                   or wire_valid is True and rejection in BINDING_CODES))
        return
    check(checks, 'expanded_compact_has_valid_raw_record_and_canonical_plan:' + rid,
          wire_valid is True and rejection is None and attempt.get('plan') is not None)
    try:
        plan = pinned.GroundedPlan.model_validate(attempt['plan'])
        wire = {'context_sha256': plan.context_sha256, 'decision': plan.decision}
        if plan.decision == 'retrieve_more':
            wire['retrieval_queries'] = list(plan.retrieval_queries)
            check(checks, 'retrieve_more_raw_observation_has_no_source_field:' + rid,
                  indices is None and count is None)
        else:
            wire['sources'] = indices
            check(checks, 'source_index_count_matches_canonical_source_count:' + rid,
                  safe_indices and len(indices) == count == len(plan.sources))
            if plan.decision == 'ask':
                wire['question_name'] = plan.question_name
        expanded = pinned.expand_compact_plan(json.dumps(wire), request)
        expansion_matches = expanded.model_dump(mode='json') == attempt['plan']
        output_hash_matches = sha(expanded.model_dump_json().encode()) == attempt.get('output_sha256')
        check(checks, 'independent_current_index_expansion_exact:' + rid, expansion_matches)
        check(checks, 'independent_expanded_canonical_output_hash:' + rid, output_hash_matches)
        check(checks, 'raw_and_expanded_hashes_have_distinct_meanings:' + rid,
              hash_shape(attempt.get('output_sha256')) and raw_hash != attempt['output_sha256'])
        check(checks, 'downstream_schema_observation_is_canonical_success:' + rid,
              attempt.get('schema_observation', {}).get('syntactic_valid') is True)
        review['canonical_expansion_independently_checked'] = expansion_matches
        review['canonical_output_hash_independently_checked'] = output_hash_matches
    except (ValueError, TypeError, KeyError, AttributeError, RecursionError):
        check(checks, 'independent_current_index_expansion_exact:' + rid, False)

def audit_controlled_bundle(checks,reviews,rid,current,bundle,job,pinned):
    version=next((f.value for f in current.case.facts if f.name.casefold()=='streamlit_version'
                  and f.status=='active' and f.provenance_kind=='user'),None)
    hits=tuple(hit for hit in bundle.hits if not hit.applicable_versions or version in hit.applicable_versions)
    expected=bundle.model_copy(update={'hits':hits,'retrieval_round':current.evidence.retrieval_round})
    matches=expected.model_dump(mode='json')==current.evidence.model_dump(mode='json')
    check(checks,'exact_registered_fixed_records_then_current_USER_scope:'+rid,matches)
    check(checks,'no_packing_clipping_or_mutated_record:'+rid,
          all(hit==next(original for original in bundle.hits if original.chunk_id==hit.chunk_id)
              for hit in current.evidence.hits))
    capacity=pinned.source_capacity(current,source_gap_handoff=True,source_gap_investigation=True)
    reviews.append({'request_id':rid,'arm':job['arm'],'case_id':job['case_id'],
                    'fixture_hits':len(bundle.hits),'actual_current_hits':len(current.evidence.hits),
                    'actual_model_bundle_reconstructed':matches,'current_user_version':version,
                    'source_capacity_advertised_bound':capacity,'whole_current_bundle_utf8_bytes':len(current.evidence.model_dump_json().encode()),
                    'guide_version_applicability':'unconfirmed; no exact1.31 inference',
                    'actual_search':False,'semantic_support_or_usefulness_certified':False})


def turn_inventory(receipt, fixtures, checks, pinned):
    """Derive coverage from all registered turns, not successful outputs alone."""
    method = receipt['method']
    batches = {batch['batch_id']: batch for batch in receipt['batches']}
    check(checks, 'actual_batch_ids_unique_and_registered',
          len(batches) == len(receipt['batches'])
          and set(batches) <= {batch['batch_id'] for batch in method['batches']})
    inventory = []
    for registered in method['batches']:
        actual = batches.get(registered['batch_id'])
        jobs = {job['job_id']: job for job in actual['jobs']} if actual else {}
        check(checks, 'actual_job_ids_unique_and_registered:' + registered['batch_id'],
              len(jobs) == (len(actual['jobs']) if actual else 0)
              and set(jobs) <= {job['job_id'] for job in registered['jobs']})
        for planned in registered['jobs']:
            job = jobs.get(planned['job_id'])
            fixture = fixtures[planned['case_id']]
            actual_steps = [step for step in job['steps'] if 'request' in step] if job else []
            steps = {step['turn']: step for step in actual_steps}
            check(checks, 'actual_turn_ids_unique_and_registered:' + planned['job_id'],
                  len(steps) == len(actual_steps)
                  and set(steps) <= {turn['id'] for turn in fixture['turns']})
            for turn in fixture['turns']:
                step = steps.get(turn['id'])
                status = 'unvisited' if step is None else 'no_fresh_proposal'
                decision = None
                proposal = step.get('proposal') if step else None
                if proposal and proposal.get('review_handoff'):
                    if (proposal.get('stale') is False
                            and proposal['case_revision'] == step['request']['expected_revision'] + 1):
                        status = 'fresh_proposal'
                        handoff = pinned.validate_handoff(
                            pinned.ProposalHandoff.model_validate(proposal['review_handoff']))
                        check(checks, 'fresh_handoff_case_revision_analysis_identity:' + planned['job_id'] + ':' + turn['id'],
                              handoff.case.case_id == proposal['case_id'] == planned['case_id']
                              and handoff.case.case_revision == proposal['case_revision']
                              and handoff.proposal_id == proposal['proposal_id']
                              and handoff.analysis.model_dump(mode='json') == proposal['analysis'])
                        decision = handoff.analysis.decision
                    else:
                        status = 'stale_observation'
                releases=step.get('released_outputs',[]) if step else []
                check(checks,'released_text_keeps_current_original_or_edited_draft:'+planned['job_id']+':'+turn['id'],
                      type(releases) is list
                      and (not releases or status=='fresh_proposal')
                      and all(output.get('role')=='assistant'
                              and output.get('decision')==(step.get('edited_proposal') or proposal)['analysis']['decision']
                              and output.get('text')==(step.get('edited_proposal') or proposal)['analysis']['user_draft']
                              for output in releases))
                inventory.append({
                    'job_id': planned['job_id'], 'case_id': planned['case_id'],
                    'round': planned['round'], 'arm': planned['arm'],
                    'stage': 'controlled_evidence', 'turn_id': turn['id'],
                    'status': status, 'decision': decision,
                    'released_output_count':len(releases),
                    'released_output_sha256':[sha(output['text'].encode()) for output in releases],
                    'edited_output_count':sum(output.get('edited') is True for output in releases),
                })
    return inventory




def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--output', type=Path, default=OUT / 'source-order-cost-review-v2.json')
    parser.add_argument('--source-commit', required=True)
    parser.add_argument('--receipt-sha256', required=True)
    args = parser.parse_args()
    if not re.fullmatch(r'[0-9a-f]{40}', args.source_commit) or not hash_shape(args.receipt_sha256):
        raise ValueError('AuditExecutionBindingRejected')
    if not args.output.resolve().is_relative_to(OUT.resolve()) or args.output.resolve().is_relative_to(ROOT.resolve()):
        raise ValueError('AuditPrivateOutputRequired')
    if args.output.exists():
        raise ValueError('AuditOutputAlreadyExists')
    checks, wire_reviews, pre_dispatch, unaccounted, native_reviews = [], [], [], [], []
    output_reviews, candidate_reviews = [], []
    raw = (PRIVATE / 'receipt.json').read_bytes()
    if sha(raw) != args.receipt_sha256:
        raise ValueError('AuditReceiptBindingRejected')
    receipt = json.loads(raw)
    method, registration = receipt['method'], receipt['registration']
    source = registration['source_commit']
    if source != args.source_commit:
        raise ValueError('AuditExecutionBindingRejected')
    pinned = pinned_helpers(source)
    method_raw = (ROOT / FOLDER / 'live-method-v1.json').read_bytes()
    check(checks, 'frozen_method_exact_embedded_bytes', sha(method_raw) == registration['method_sha256']
          and method_raw == (json.dumps(method, ensure_ascii=False, indent=2) + '\n').encode())
    check(checks, 'source_registration_and_end_match', source == receipt['source_after']['source_commit']
          and registration['source_hashes'] == receipt['source_after']['source_hashes']
          and receipt['source_unchanged'] is True)
    check(checks, 'registration_is_original_file', json.loads((PRIVATE / 'registration.json').read_bytes()) == registration)
    frozen_inputs = {}
    for name, pin in {**registration['source_hashes'], **method['input_pins']}.items():
        archived = subprocess.check_output(['git', 'show', f'{source}:{name}'], cwd=ROOT)
        frozen_inputs[name] = archived
        check(checks, 'paid_commit_pin:' + name, sha(archived) == pin)
    for name, pin in registration['source_hashes'].items():
        if name.startswith('src/agent_app/'):
            check(checks, 'wire_audit_helper_loaded_from_paid_source:' + name,
                  sha((pinned.root / name).read_bytes()) == pin)
    check(checks,'controlled_evidence_campaign_identity',
          method['protocol_id']==receipt['protocol_id']=='core-source-order-model-v1'
          and method['experiment_id']=='core-source-order-2026-10-15-v1'
          and method['authorization_ref']=='user-2026-10-15-complementary-selection-order-diagnostic')
    check(checks,'exact6pairs12jobs24turns48freshslots',len(method['batches'])==6
          and sum(len(b['jobs']) for b in method['batches'])==12
          and len(method['fresh_request_ids'])==len(set(method['fresh_request_ids']))==48
          and method['scope_cap_usd']=='0.06' and method['campaign_scope_cap_usd']=='0.36')
    check(checks,'no_real_RAG_or_final_quality_claim',method['real_RAG_retrieval'] is False
          and receipt['real_RAG_retrieval'] is False
          and method['controlled_fixed_evidence'] is True and receipt['controlled_fixed_evidence'] is True
          and method['answer_linked_guide_fixture_only'] is True and receipt['answer_linked_guide_fixture_only'] is True
